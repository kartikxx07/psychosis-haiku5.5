from scipy.stats import kstest
import numpy as np

# Example transaction amounts
transaction_amounts = [100, 161.8, 261.8, 423.6, 685.4]

# Perform Kolmogorov-Smirnov test against a Fibonacci sequence
def fibonacci_sequence(n):
    fib = [0, 1]
    for i in range(2, n):
        fib.append(fib[-1] + fib[-2])
    return fib

# Generate Fibonacci sequence up to the maximum transaction amount

# Normalize the sequences

ks_statistic, p_value = kstest(normalized_transactions, normalized_fibonacci)
